      }

      dotBatt.classList.remove('flow-dot-active-batt', 'flow-dot-reverse-batt');

      if (chargeCurrent >= 0.1) {

      } else if (chargeCurrent === 0 && loadPercentage >= 1) {

      async function syncTimeMPPT(e) {
        if (e) e.preventDefault(); // Mencegah perilaku default elemen <a> (reload/jump ke atas)

        try {
          const response = await fetch('/sync', {
            method: 'POST'
          });

          if (response.ok) {
            alert("Berhasil! Waktu telah disinkronkan ke MPPT.");
          } else {
            const errorText = await response.text();
            alert("Gagal: " + errorText);
          }
        } catch (error) {
          console.error("Error syncing time:", error);
          alert("Terjadi kesalahan koneksi ke perangkat ESP!");
        } finally {
          // Menutup menu dropdown secara otomatis setelah diklik
          settingsMenu.classList.add('hidden');
        }

        })
        .catch(err => console.error("Gagal memuat kapasitas PV:", err));
    });

    // Tutup Modal PV
    closePvModal.addEventListener('click', () => pvModal.classList.add('hidden'));
    pvModal.addEventListener('click', (e) => {
      if (e.target === pvModal) pvModal.classList.add('hidden');
    });

    // Simpan Nilai ke ESP8266 & EEPROM
    pvForm.addEventListener('submit', async (e) => {
      e.preventDefault();
      const value = parseFloat(pvInput.value);
      if (isNaN(value) || value <= 0) return;

      try {
        const res = await fetch('/pv-capacity', {
          method: 'POST',
          headers: { 'Content-Type': 'text/plain' },
          body: value.toString()
        });

        if (res.ok) {
          alert('Kapasitas Panel Surya berhasil disimpan!');
          pvModal.classList.add('hidden');
          updateData();
        } else {
          alert('Gagal menyimpan konfigurasi!');
        }
      } catch (err) {
        console.error(err);
        alert('Terjadi kesalahan koneksi!');
      }
    });

    function updateDateTime() {
      const now = new Date();

      // Format Tanggal (Contoh: 22 Agt 2026)
      const dateOptions = { day: 'numeric', month: 'short', year: 'numeric' };
      document.getElementById('currentDate').textContent = now.toLocaleDateString('id-ID', dateOptions);

      // Format Waktu (24 Jam: 15:04:05)
      const timeOptions = { hour: '2-digit', minute: '2-digit', second: '2-digit', hour12: false };
      document.getElementById('currentTime').textContent = now.toLocaleTimeString('id-ID', timeOptions);

      // Format Offset UTC (Contoh: UTC+7)
      const offsetHours = -now.getTimezoneOffset() / 60;
      const sign = offsetHours >= 0 ? '+' : '';
      document.getElementById('utcOffset').textContent = `(UTC${sign}${offsetHours})`;
    }

    // Update langsung & jalankan interval per 1 detik
    updateDateTime();
    setInterval(updateDateTime, 1000);

    updateData();
    setInterval(updateData, 3000);
